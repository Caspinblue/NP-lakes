    "import os\n",
    "import re\n",
    "import time\n",

    "import numpy as np\n",

    "def parse_datetime_force2024(s):\n",
    "    if pd.isna(s) or not isinstance(s, str) or s.strip() == \"\" or s.strip() == \"?\":\n",
    "        return pd.NaT\n",
    "    s = s.strip()\n",
    "    parts = s.split()\n",
    "    if len(parts) == 1:\n",
    "        date_part = parts[0]\n",
    "        time_part = \"00:00\"\n",
    "    else:\n",
    "        date_part, time_part = parts[0], parts[1]\n",
    "\n",
    "    date_fields = re.split(r\"[/-]\", date_part)\n",
    "    date_fields = [x for x in date_fields if x != \"\"]\n",
    "    if len(date_fields) < 2:\n",
    "        return pd.NaT\n",
    "    if len(date_fields) >= 3:\n",
    "        month = date_fields[-2]\n",
    "        day = date_fields[-1]\n",
    "    else:\n",
    "        month, day = date_fields[0], date_fields[1]\n",
    "\n",
    "    month = month.zfill(2)\n",
    "    day = day.zfill(2)\n",
    "\n",
    "    time_fields = time_part.split(\":\")\n",
    "    if len(time_fields) >= 2:\n",
    "        hour = time_fields[0].zfill(2)\n",
    "        minute = time_fields[1].zfill(2)\n",
    "    else:\n",
    "        hour, minute = \"00\", \"00\"\n",
    "    second = \"00\"\n",
    "\n",
    "    datetime_str = f\"2024-{month}-{day} {hour}:{minute}:{second}\"\n",
    "    return pd.to_datetime(datetime_str, format=\"%Y-%m-%d %H:%M:%S\", errors=\"coerce\")\n",
    "\n",
    "\n",
    "def get_model(model_name):\n",
    "    if model_name == \"ETR\":\n",
    "        return ExtraTreesRegressor(n_estimators=100, random_state=42, n_jobs=-1)\n",
    "    elif model_name == \"RFR\":\n",
    "        return RandomForestRegressor(n_estimators=100, random_state=42, n_jobs=-1)\n",
    "    elif model_name == \"XGB\":\n",
    "        return XGBRegressor(n_estimators=100, random_state=42, n_jobs=-1)\n",
    "    elif model_name == \"LGBM\":\n",
    "        return LGBMRegressor(n_estimators=100, random_state=42, n_jobs=-1, verbose=-1)\n",
    "    elif model_name == \"KNN\":\n",
    "        return KNeighborsRegressor(n_neighbors=5)\n",
    "    elif model_name == \"DTR\":\n",
    "        return DecisionTreeRegressor(random_state=42)\n",
    "    elif model_name == \"SVR\":\n",
    "        return SVR(kernel=\"rbf\")\n",
    "    else:\n",
    "        raise ValueError(f\"Unknown model: {model_name}\")\n",
    "\n",
    "\n",
    "def calc_metrics(y_true, y_pred):\n",
    "    r2 = round(r2_score(y_true, y_pred), 4)\n",
    "    rmse = round(np.sqrt(mean_squared_error(y_true, y_pred)), 4)\n",
    "    mre = round(np.mean(np.abs((y_true - y_pred) / y_true)), 4)\n",
    "    return r2, rmse, mre\n",
    "\n",
    "\n",
    "def mre_score_func(y_true, y_pred):\n",
    "    return np.mean(np.abs((y_true - y_pred) / y_true))\n",
    "\n",
    "\n",
    "mre_scorer = make_scorer(mre_score_func, greater_is_better=False)\n",
    "\n",
    "\n",
    "def make_stratify_groups(months, min_samples=2):\n",
    "    month_counts = months.value_counts().sort_index()\n",
    "    groups = []\n",
    "    current_group = []\n",
    "    current_count = 0\n",
    "    for month in month_counts.index:\n",
    "        count = month_counts[month]\n",
    "        if current_count == 0:\n",
    "            current_group = [month]\n",
    "            current_count = count\n",
    "        else:\n",
    "            if current_count < min_samples:\n",
    "                current_group.append(month)\n",
    "                current_count += count\n",
    "            else:\n",
    "                groups.append(current_group)\n",
    "                current_group = [month]\n",
    "                current_count = count\n",
    "    if current_group:\n",
    "        groups.append(current_group)\n",
    "\n",
    "    month_to_group = {}\n",
    "    for i, group in enumerate(groups):\n",
    "        for m in group:\n",
    "            month_to_group[m] = i\n",
    "    return months.map(month_to_group)\n",
    "\n",
    "\n",
    "def process_folder(input_dir, output_dir, cv_folds=5):\n",
    "    train_dir = os.path.join(output_dir, \"train\")\n",
    "    test_dir = os.path.join(output_dir, \"test\")\n",

    "        \"Water Temperature(℃)\",\n",
    "        \"Electrical Conductivity(μS/cm)\",\n",
    "        \"Turbidity(NTU)\",\n",
    "        \"Ammonia Nitrogen(mg/L)\",\n",
    "        \"pH(Dimensionless)\",\n",
    "        \"Dissolved Oxygen(mg/L)\",\n",
    "        \"Chlorophyllα(mg/L)\",\n",
    "    ]\n",
    "    targets = [\"Total Nitrogen(mg/L)\", \"Total Phosphorus(mg/L)\"]\n",
    "\n",
    "    model_names = [\"ETR\", \"RFR\", \"XGB\", \"LGBM\", \"KNN\", \"DTR\", \"SVR\"]\n",
    "\n",
    "    model_total_time = {name: 0.0 for name in model_names}\n",
    "    model_file_count = {name: 0 for name in model_names}\n",
    "\n",
    "    for file_name in csv_files:\n",

    "\n",
    "        df[\"Parsed_Time\"] = df[\"Monitoring Time\"].apply(parse_datetime_force2024)\n",
    "        df[\"Month\"] = df[\"Parsed_Time\"].dt.month\n",
    "\n",
    "        df_normal = df[df[\"Station Status\"] == \"Normal\"].copy()\n",
    "        for col in feature_cols + targets:\n",
    "            df_normal[col] = pd.to_numeric(df_normal[col], errors=\"coerce\")\n",
    "        df_clean = df_normal.dropna(subset=feature_cols + targets).reset_index(drop=True)\n",
    "        df_clean = df_clean.dropna(subset=[\"Month\"]).reset_index(drop=True)\n",
    "        df_clean[\"Month\"] = df_clean[\"Month\"].astype(int)\n",
    "\n",
    "        if len(df_clean) < 10:\n",
    "            continue\n",
    "\n",
    "        month_counts = df_clean[\"Month\"].value_counts()\n",
    "        if (month_counts < 2).any():\n",
    "            stratify_groups = make_stratify_groups(df_clean[\"Month\"], min_samples=2)\n",
    "            group_counts = stratify_groups.value_counts()\n",
    "            if (group_counts < 2).any():\n",
    "                use_stratify = False\n",
    "            else:\n",
    "                use_stratify = True\n",
    "        else:\n",
    "            stratify_groups = df_clean[\"Month\"]\n",
    "            use_stratify = True\n",
    "\n",
    "        if use_stratify:\n",
    "            idx_train, idx_test = train_test_split(\n",
    "                np.arange(len(df_clean)), test_size=0.25, stratify=stratify_groups, random_state=42\n",
    "            )\n",
    "        else:\n",
    "            idx_train, idx_test = train_test_split(np.arange(len(df_clean)), test_size=0.25, random_state=42)\n",

    "        train_data_raw = df_clean.iloc[idx_train].copy()\n",
    "        test_data_raw = df_clean.iloc[idx_test].copy()\n",
    "\n",
    "        train_file = os.path.join(train_dir, f\"{base_name}_train.csv\")\n",
    "        train_data_raw.to_csv(train_file, index=False, encoding=\"utf-8-sig\")\n",
    "\n",
    "        scores_list = []\n",
    "\n",
    "        for model_name in model_names:\n",
    "            model_start = time.perf_counter()\n",
    "\n",
    "            try:\n",
    "                model_tn = get_model(model_name)\n",
    "                X_train = train_data_raw[feature_cols]\n",
    "                y_train_tn = train_data_raw[targets[0]]\n",
    "                X_test = test_data_raw[feature_cols]\n",
    "                y_test_tn = test_data_raw[targets[0]]\n",
    "\n",
    "                cv_results_tn = cross_validate(\n",
    "                    model_tn,\n",
    "                    X_train,\n",
    "                    y_train_tn,\n",

    "                cv_r2_mean = round(np.mean(cv_results_tn[\"test_r2\"]), 4)\n",

    "                        \"rmse\": \"neg_root_mean_squared_error\",\n",
    "                        \"mre\": mre_scorer,\n",
    "                    },\n",
    "                    return_train_score=False,\n",
    "                    n_jobs=-1,\n",
    "                )\n",
    "                cv_r2_mean_tp = round(np.mean(cv_results_tp[\"test_r2\"]), 4)\n",
    "                cv_r2_std_tp = round(np.std(cv_results_tp[\"test_r2\"]), 4)\n",
    "                cv_rmse_mean_tp = round(np.mean(-cv_results_tp[\"test_rmse\"]), 4)\n",
    "                cv_rmse_std_tp = round(np.std(-cv_results_tp[\"test_rmse\"]), 4)\n",
    "                cv_mre_mean_tp = round(np.mean(-cv_results_tp[\"test_mre\"]), 4)\n",
    "                cv_mre_std_tp = round(np.std(-cv_results_tp[\"test_mre\"]), 4)\n",
    "\n",
    "                model_tp.fit(X_train, y_train_tp)\n",
    "                y_pred_tp = model_tp.predict(X_test)\n",
    "                y_pred_tp = np.round(y_pred_tp, 4)\n",
    "\n",
    "                test_r2_tp, test_rmse_tp, test_mre_tp = calc_metrics(y_test_tp.values, y_pred_tp)\n",
